    #!/usr/bin/env python3 """ create_image.py Place your background image named `image.png` in the same folder, then run: pip install pillow python create_image.py This script writes the project content onto the image and saves image_with_projects.png """ from PIL import Image, ImageDraw, ImageFont import textwrap import os # Configuration INPUT_IMAGE = "image.png" OUTPUT_IMAGE = "image_with_projects.png" FONT_PATH = None # e.g., "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf" FONT_SIZE = 22 LINE_SPACING = 6 MARGIN = 40 RECT_PADDING = 20 TEXT_COLOR = (255, 255, 255, 255) # white RECT_COLOR = (0, 0, 0, 180) # semi-transparent black # The project text to write (you can change formatting or bullets) project_text = ( "E-Commerce Data Analysis:\n" "- At 4668, Tulika had the highest Sum of Amount and was 15,460.00% higher than Jaydeep (lowest: 30).\n" "- Tulika: 4668 (highest), followed by Abhijeet and Anurag. Jaydeep: 30 (lowest).\n" "- Tulika accounted for 5.34% of Sum of Amount.\n" "- Madhya Pradesh had 87463 Sum of Amount.\n\n" "HR Data Analysis:\n" "- 50 total employees, mostly male.\n" "- IT leads in salary; training is periodic and intense.\n" "- Tools: Power BI, Excel. Charts: Pie, Donut, Bar, Line, Map.\n\n" "Hospital Data Analysis:\n" "- 'Other' had highest total Sum of Age at 3548, Female 3023, Male 2853.\n" "- 'Other' made up 7.34% of Sum of Age.\n" "- All specializations had Count of Head_of_Department = 10.\n\n" "Air Quality Report Analysis:\n" "- Average AQI trended down; 78.86% decrease between Jan 2015 and Jul 2020.\n" "- AQI started trending down March 2019, falling 57.04% in 16 months.\n" "- Steepest decline: 343 -> 298.16 between Jan 2015 and Mar 2015.\n" "- Coimbatore had highest average AQI 73.02; Aizawl lowest 34.77; Shillong 53.80." ) def find_font(font_size): # Try common fonts candidates = [ FONT_PATH, "/usr/share/fonts/truetype/dejavu/DejaVuSans-Bold.ttf", "/usr/share/fonts/truetype/dejavu/DejaVuSans.ttf", "/Library/Fonts/Arial.ttf", "C:\\Windows\\Fonts\\arial.ttf", ] for p in candidates: if not p: continue if os.path.isfile(p): try: return ImageFont.truetype(p, font_size) except Exception: continue # fallback to default PIL font return ImageFont.load_default() def wrap_text(text, font, max_width, draw): lines = [] for paragraph in text.split("\n"): if paragraph.strip() == "": lines.append("") continue # Wrap paragraph wrapped = textwrap.wrap(paragraph, width=100) # start with large wrap # refine by measuring new_wrapped = [] for line in wrapped: # further split long measured lines words = line.split() current = "" for w in words: test = current + (" " if current else "") + w w_width, _ = draw.textsize(test, font=font) if w_width <= max_width: current = test else: if current: new_wrapped.append(current) current = w if current: new_wrapped.append(current) lines.extend(new_wrapped) return lines def main(): if not os.path.isfile(INPUT_IMAGE): print(f"Input image '{INPUT_IMAGE}' not found. Place your background image named '{INPUT_IMAGE}' in this folder.") return im = Image.open(INPUT_IMAGE).convert("RGBA") W, H = im.size print(f"Loaded {INPUT_IMAGE} ({W}x{H})") txt_layer = Image.new("RGBA", im.size, (255,255,255,0)) draw = ImageDraw.Draw(txt_layer) font = find_font(FONT_SIZE) # Determine text area max width (leave margins) max_text_width = W - 2 * MARGIN - 2 * RECT_PADDING # Wrap text into lines lines = wrap_text(project_text, font, max_text_width, draw) # Compute height required line_height = font.getsize("Ay")[1] + LINE_SPACING text_block_height = line_height * len(lines) # Position text block near bottom-left, or change as needed x = MARGIN y = H - MARGIN - text_block_height # Draw semi-transparent rectangle behind text rect_x0 = x - RECT_PADDING rect_y0 = y - RECT_PADDING rect_x1 = x + max_text_width + RECT_PADDING rect_y1 = y + text_block_height + RECT_PADDING # Constrain rect to image rect_x1 = min(rect_x1, W - MARGIN) rect_y1 = min(rect_y1, H - MARGIN) draw.rectangle([(rect_x0, rect_y0), (rect_x1, rect_y1)], fill=RECT_COLOR) # Draw text lines cur_y = y for line in lines: draw.text((x, cur_y), line, font=font, fill=TEXT_COLOR) cur_y += line_height # Composite text layer onto image out = Image.alpha_composite(im, txt_layer) out.convert("RGB").save(OUTPUT_IMAGE, "PNG") print(f"Saved {OUTPUT_IMAGE}") if __name__ == "__main__": main()
